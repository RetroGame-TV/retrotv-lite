cmake_minimum_required(VERSION 3.22)
project(retrotv LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(RETROTV_TESTS "Build portable policy tests without SDL/Android" OFF)
if(RETROTV_TESTS)
  enable_testing()
  add_executable(snes_save_tests tests/snes_save_tests.cpp)
  target_include_directories(snes_save_tests PRIVATE .)
  add_test(NAME snes_save_preservation COMMAND snes_save_tests ${CMAKE_BINARY_DIR}/snes-saves-$<CONFIG>)
  add_executable(policy_tests tests/policy_tests.cpp)
  target_include_directories(policy_tests PRIVATE .)
  add_test(NAME policy_tests COMMAND policy_tests)
  include(../cores/quicknes.cmake)
  add_executable(host_tests tests/host_tests.cpp emulation/host.cpp)
  target_include_directories(host_tests PRIVATE . ../third_party/quicknes/libretro/libretro-common/include)
  if(UNIX)
    target_link_libraries(host_tests dl)
  endif()
  add_test(NAME real_quicknes COMMAND host_tests $<TARGET_FILE:quicknes_libretro> ${CMAKE_CURRENT_SOURCE_DIR}/../assets/demo/retrotv-pad-test.nes ${CMAKE_BINARY_DIR}/test-saves)
  add_executable(vfs_tests tests/vfs_tests.cpp)
  target_include_directories(vfs_tests PRIVATE . ../third_party/quicknes/libretro/libretro-common/include)
  target_compile_definitions(vfs_tests PRIVATE _FILE_OFFSET_BITS=64)
  add_test(NAME large_random_access COMMAND vfs_tests ${CMAKE_BINARY_DIR}/test-vfs)
  # Exercise the actual SDL virtual-controller API and the sampler used by main.cpp, without a window/audio.
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
  set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
  set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
  set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
  add_subdirectory(../third_party/SDL ${CMAKE_BINARY_DIR}/sdl)
  add_executable(frame_input_tests tests/frame_input_tests.cpp emulation/host.cpp)
  target_include_directories(frame_input_tests PRIVATE . ../third_party/quicknes/libretro/libretro-common/include)
  target_link_libraries(frame_input_tests SDL2)
  if(UNIX)
    target_link_libraries(frame_input_tests dl)
  endif()
  target_compile_definitions(frame_input_tests PRIVATE SDL_MAIN_HANDLED)
  if(WIN32)
    add_custom_command(TARGET frame_input_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2> $<TARGET_FILE_DIR:frame_input_tests>)
  endif()
  add_test(NAME late_frame_controller_input COMMAND frame_input_tests $<TARGET_FILE:quicknes_libretro> ${CMAKE_CURRENT_SOURCE_DIR}/../assets/demo/retrotv-pad-test.nes)
  return()
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(../third_party/SDL ${CMAKE_BINARY_DIR}/sdl)
option(RETROTV_PACKAGED_QUICKNES "Use an explicitly supplied core in cores/packaged for a local update" OFF)
if(NOT RETROTV_PACKAGED_QUICKNES)
  include(../cores/quicknes.cmake)
endif()
add_library(main SHARED main.cpp emulation/host.cpp)
target_include_directories(main PRIVATE . ../third_party/quicknes/libretro/libretro-common/include)
target_compile_definitions(main PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(main SDL2 GLESv2 android log dl)
